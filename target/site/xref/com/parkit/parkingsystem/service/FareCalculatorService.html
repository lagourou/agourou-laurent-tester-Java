13      public void calculateFare(Ticket ticket, boolean discount) {
14  
15          if ((ticket.getOutTime() == null) || (ticket.getOutTime().before(ticket.getInTime()))) {
16              throw new IllegalArgumentException("Out time provided is incorrect:" + ticket.getOutTime().toString());
17          }
18  
19          long inTime = ticket.getInTime().getTime();
20          long outTime = ticket.getOutTime().getTime();
21  
22          // TODO: Some tests are failing here. Need to check if this logic is correct
23  
24          long duration = outTime - inTime;
25          double minutes = duration / (1000.0 * 60);
26          double hours = minutes / 60.0;
27  
28          if (minutes <= 30) {
29              ticket.setPrice(0);
30              return;
31          }
32          double rate;
33  
34          switch (ticket.getParkingSpot().getParkingType()) {
35              case CAR: {
36                  rate = Fare.CAR_RATE_PER_HOUR;
37                  ticket.setPrice(hours * rate);
38                  break;
39              }
40              case BIKE: {
41                  rate = Fare.BIKE_RATE_PER_HOUR;
42                  ticket.setPrice(hours * rate);
43                  break;
44              }
45              default:
46                  throw new IllegalArgumentException("Unkown Parking Type");
47          }
48          if (discount) {
49              ticket.setPrice(ticket.getPrice() * 0.95);
50          }
51      }
52  
53      public void calculateFare(Ticket ticket) {
54          calculateFare(ticket, false);
55      }
